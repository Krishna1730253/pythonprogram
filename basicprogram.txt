"""Program 1.17: check a prime and list primes."""
def is_prime(number):
    if number < 2:
        return False
    for divisor in range(2, int(number ** 0.5) + 1):
        if number % divisor == 0:
            return False
    return True

number = int(input("Enter N: "))
if is_prime(number):
    print(number, "is prime.")
else:
    print(number, "is not prime.")
print("Primes from 2 to N:", *[n for n in range(2, number + 1) if is_prime(n)])
'''
Enter N: 6
6 is not prime.
Primes from 2 to N: 2 3 5
'''

"""Program 1.19: reverse and test a number."""
original = int(input("Enter a non-negative number: "))
if original < 0:
    print("Please enter a non-negative number.")
else:
    number, reverse = original, 0
    while number > 0:
        reverse = reverse * 10 + number % 10
        number //= 10
    print("Reversed number:", reverse)
    print("It is a palindrome." if original == reverse else "It is not a palindrome.")
'''
Enter a non-negative number: 5
Reversed number: 5
It is a palindrome.

"""Program 1.21: Armstrong number checker and finder."""
def is_armstrong(number):
    digits = str(number)
    return number == sum(int(digit) ** len(digits) for digit in digits)

number = int(input("Enter a number: "))
if is_armstrong(number):
    print(number, "is an Armstrong number.")
else:
    print(number, "is not an Armstrong number.")
print("Armstrong numbers from 1 to 999:")
print(*[n for n in range(1, 1000) if is_armstrong(n)])
'''
Enter a number: 12
12 is not an Armstrong number.
Armstrong numbers from 1 to 999:
1 2 3 4 5 6 7 8 9 153 370 371 407
'''

"""Program 2.4: multi-return statistics function."""
def stats(a, b, c):
    return a + b + c, (a + b + c) / 3, min(a, b, c), max(a, b, c)

"""Program 2.5: reverse strings and check palindromes."""
def reverse_string(text): return text[::-1]
def is_palindrome(text): return text.lower() == reverse_string(text).lower()

samples = ["madam", "racecar", "Your Name", "Your City"]
for text in samples:
    print(text, ": reverse =", reverse_string(text), ", palindrome =", is_palindrome(text))
'''
madam : reverse = madam , palindrome = True
racecar : reverse = racecar , palindrome = True
Your Name : reverse = emaN ruoY , palindrome = False
Your City : reverse = ytiC ruoY , palindrome = False
'''

"""Program 2.6: vowel and character frequency functions."""
def count_vowels(text): return sum(char.lower() in "aeiou" for char in text)
def char_frequency(text):
    counts = {}
    for char in text: counts[char] = counts.get(char, 0) + 1
    return counts

sentence = input("Enter a sentence: ")
print("Vowels:", count_vowels(sentence))
print("Character frequency:", char_frequency(sentence))
'''
Enter a sentence: how are you?
Vowels: 5
Character frequency: {'h': 1, 'o': 2, 'w': 1, ' ': 2, 'a': 1, 'r': 1, 'e': 1, 'y': 1, 'u': 1, '?': 1}
'''
