247:login

user_id = input("Enter User ID: ").strip()
user_pass = input("Enter Password: ").strip()

found = False


    
file = open("users.txt", "r")
    
    
for line in file:
        data = line.strip().split(",")
        stored_user = data[0]
        stored_pass = data[1]
        
        
        if user_id == stored_user and user_pass == stored_pass:
            found = True
            break
            
file.close()

if found:
        print("Login Successful!")
else:
        print("Invalid User ID or Password.")

# Write data separated by comma with a newline at the end
file.write(username + "," + password + "\n")

# Close the file
file.close()
